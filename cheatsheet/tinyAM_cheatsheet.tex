\smallkey{Start with a deliberately simple baseline}

base <- fit_tam(
  dat, years = 1983:2024, ages = 2:14,
  N_settings   = list(process = "off", init = "exp"),
  rec_settings = list(process = "iid", mu_form = ~ 1),
  F_settings   = list(process = "iid", mu_form = ~ 1),
  M_settings   = list(process = "off",
                      mu_supplied = ~ I(0.3)),

\pageheader{tinyAM :: MODEL MATHEMATICS}{02 / STATES + PROCESSES + OBSERVATIONS\hspace{2em} | \hspace{.5em} Compact mathematical reference for the model core}
\begin{minipage}[t]{.325\linewidth}\vspace{0pt}
\begin{mathbox}{Age-structured states}\micro{Years $t=1,\ldots,T$; ages $a=1,\ldots,A$; $A$ is the terminal plus group. $N_{t,a}$ is start-of-year abundance; $F$ and $M$ are annual instantaneous mortalities.}
\[
Z_{t,a}=F_{t,a}+M_{t,a}, \qquad R_t=N_{t,1}.
\]
\end{mathbox}\vspace{2mm}
\begin{mathbox}{Cohort survival}\micro{Expected survivors advance to the next age:}
\[
\widehat N_{t,a}=N_{t-1,a-1}\exp(-Z_{t-1,a-1}),\quad 2\le a<A.
\]
\micro{The plus group also retains surviving fish already at age $A$:}
\[
\begin{aligned}
\widehat N_{t,A}={}&N_{t-1,A-1}e^{-Z_{t-1,A-1}}\\
 &+N_{t-1,A}e^{-Z_{t-1,A}}.
\end{aligned}
\]
\micro{With an active abundance process:}
\[
\log N_{t,a}=\log\widehat N_{t,a}+\eta^N_{t,a}.
\]
\end{mathbox}\vspace{2mm}
\begin{mathbox}{Spawning-stock biomass}\micro{$W$: weight; $P$: mature proportion; $s^{\mathrm{spawn}}$: fraction of the year to spawning. Defaults to zero.}
\[
SSB_t=\sum_a N_{t,a}W_{t,a}P_{t,a}\,
 e^{-s^{\mathrm{spawn}}_{t,a}Z_{t,a}}.
\]
\micro{Shortened plus groups may retain older input biology through effective $W$ and $P$. Units are not converted automatically.}
\end{mathbox}\vspace{2mm}
\begin{mathbox}{Initial older-age abundance}\micro{\texttt{N\_settings\$init}: \texttt{"exp"} uses first-year cohort survivorship, \texttt{"free"} estimates older-age states, \texttt{"random"} adds IID survivorship deviations. Recruitment at $t=1$ is a free fixed anchor.}
\end{mathbox}
\end{minipage}\hfill
\begin{minipage}[t]{.325\linewidth}\vspace{0pt}
\begin{mathbox}{Recruitment dynamics}\micro{Default temporal recruitment random walk:}
\[
\log R_t=\log R_{t-1}+\epsilon^R_t,\quad
\epsilon^R_t\sim\mathcal N(0,\sigma_R^2).
\]
\micro{Formula-based means allow environmental covariates and stock--recruitment relationships:}
\[
\log R_t=\log g(SSB_{t-L})+X_t\gamma+u_t.
\]
\[
\begin{aligned}
g_{BH}(S)&=\frac{\alpha S}{1+\beta S},\\
g_{Ricker}(S)&=\alpha S e^{-\beta S}.
\end{aligned}
\]
\micro{The BH/Ricker curves require IID or AR1 recruitment residuals; RW + curve is not supported.}
\end{mathbox}\vspace{2mm}
\begin{mathbox}{Process variation}\micro{For $\eta$ in N, F or M, process choices encode different hypotheses. Component-specific ages/blocks and SD parameters apply.}
\[
\begin{aligned}
\mathrm{IID:}\quad &\eta_{t,a}\sim\mathcal N(0,\sigma_a^2),\\
\mathrm{RW:}\quad &\eta_{t,a}-\eta_{t-1,a}\sim\mathcal N(0,\sigma_a^2),\\
\mathrm{AR1:}\quad &\operatorname{Cov}(\eta_{t,a},\eta_{s,b})\\
 &=\frac{\sigma^2\,\phi_t^{|t-s|}\phi_a^{|a-b|}}
 {(1-\phi_t^2)(1-\phi_a^2)}.
\end{aligned}
\]
\micro{F also offers correlated age-specific RW increments with $\Sigma_{ab}=\sigma_a\sigma_b\rho^{|a-b|}$. The RW starting level is unpenalized.}
\end{mathbox}\vspace{2mm}
\begin{mathbox}{Mortality mean structure}\[
\log\mu^F=X_F\beta_F,\qquad
\log\mu^M=\log M^{\mathrm{supplied}}+X_M\beta_M.
\]
\micro{Where fitted, realized log-F/M states depart from these mean surfaces; \texttt{mono()} or \texttt{logistic()} can describe rising F mean curves under IID/AR1 residuals.}
\end{mathbox}
\end{minipage}\hfill
\begin{minipage}[t]{.325\linewidth}\vspace{0pt}
\begin{mathbox}{Catch and survey predictions}\micro{Baranov catch equation (numbers at age):}
\[
\widetilde C_{t,a}=N_{t,a}\frac{F_{t,a}}{Z_{t,a}}
 (1-e^{-Z_{t,a}}).
\]
\micro{Survey index at within-year sampling time $s_i\in[0,1]$:}
\[
\widetilde I_i=q_i N_{t_i,a_i}e^{-s_iZ_{t_i,a_i}}.
\]
\micro{Default catchability link: $q_i=\exp(\eta_{q,i})$; optional logit link confines $q_i$ to $(0,1)$. With \texttt{mono(q\_block)}, successive increments in $\eta_q$ are nonnegative.}
\end{mathbox}\vspace{2mm}
\begin{mathbox}{Observation distributions}\micro{Observed positive catch and index values have lognormal likelihood contributions:}
\[
\log Y_i\sim\mathcal N(\log\widetilde Y_i,\tau_i^2),
\]
\[
\tau_i=\tau_i^{\mathrm{supplied}}\exp(X_{sd,i}\beta_{sd}).
\]
\micro{$\widetilde Y$ is a conditional median. Zero/NA observations are treated as missing. Observation SD is distinct from estimation SE.}
\end{mathbox}\vspace{2mm}
\begin{mathbox}{Likelihood and inference}\micro{The joint likelihood combines recruitment, active N/F/M processes, random initial abundance and observation densities:}
\[
\ell(\theta,\eta)=\ell_R+\ell_N+\ell_F+
\ell_M+\ell_{N_0}+\ell_{\mathrm{obs}}.
\]
\micro{\texttt{fit\_tam()} fits by RTMB Laplace integration of random effects and \texttt{nlminb()} optimization. Reported uncertainty is a local curvature/delta-method approximation.}
\end{mathbox}\vspace{2mm}
\begin{mathbox}{Assess predictive skill}\micro{Retrospective peels assess stability (Mohn's $\rho$). Hindcasts hold out terminal observations and use one-step-ahead projection errors; \texttt{fit\_retro()} and \texttt{fit\_hindcast()} return summary statistics.}
\end{mathbox}
\end{minipage}
