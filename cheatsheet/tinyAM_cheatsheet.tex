\pageheader{tinyAM :: MODEL REFERENCE}{R specifications and the population assumptions they represent}
\begin{minipage}[t]{.325\linewidth}\vspace{0pt}
\begin{mathbox}{Abundance and cohort survival}
\begin{rcode}
N_settings = list(
  process = "iid", init = "exp")
\end{rcode}
\micro{$Z=F+M$; recruitment $R_t=N_{t,1}$. Age indices $1,\ldots,A$ run from the youngest modeled age to the plus group; N is start-of-year abundance.}
\[
\begin{aligned}
\widehat N_{t,a}&=N_{t-1,a-1}e^{-Z_{t-1,a-1}},\\[-1mm]
\widehat N_{t,A}&=N_{t-1,A-1}e^{-Z_{t-1,A-1}}\\[-1mm]
&\quad+N_{t-1,A}e^{-Z_{t-1,A}},\\
\log N_{t,a}&=\log\widehat N_{t,a}+\eta^N_{t,a}.
\end{aligned}
\]
\micro{For $t\ge2$, non-plus ages use the first survival equation. Initial older ages use \texttt{"exp"} survivorship, \texttt{"free"} fixed states, or \texttt{"random"} IID survivorship errors.}
\end{mathbox}\vspace{2mm}
\begin{mathbox}{Recruitment}
\begin{rcode}
rec_settings = list(process = "rw")
# Or a stock-recruit curve:
rec_settings = list(process="iid", mu_form=~bh(ssb))
\end{rcode}
\[
\begin{aligned}
\text{RW: }\log R_t-\log R_{t-1}&=\epsilon_t,\\
\log R_t&=\log g(SSB_{t-L})+u_t,\\
g_{BH}(S)&=\alpha S/(1+\beta S),\\
g_{Ricker}(S)&=\alpha S e^{-\beta S}.
\end{aligned}
\]
\micro{$\alpha,\beta>0$; $\epsilon_t\sim\mathcal N(0,\sigma_R^2)$. IID has $u_t=\epsilon_t$; AR1 has $u_t=\phi_Ru_{t-1}+\epsilon_t$ and a stationary start. Curves describe medians. Lag $L$ defaults to the youngest biological age; early boundary R is free. No RW + curve.}
\end{mathbox}\vspace{2mm}
\begin{mathbox}{Biomass and spawning biomass}
\begin{rcode}
dat$maturity$spawn_time <- 0.25
\end{rcode}
\[
SSB_t=\sum_b n_{t,b}w_{t,b}p_{t,b}e^{-\tau_{t,b}Z_{t,b}}.
\]
\micro{Biomass is $\sum_b n_{t,b}w_{t,b}$. $b$: retained biological age; $n$: its abundance; $w$: weight; $p$: mature fraction; $\tau$: year fraction (default 0). Hidden ages retain biology/timing and share plus-age mortality.}
\end{mathbox}
\end{minipage}\hfill
\begin{minipage}[t]{.325\linewidth}\vspace{0pt}
\begin{mathbox}{Mean fishing mortality}
\begin{rcode}
F_settings = list(process = "iid",
  mu_form = ~ logistic(age))
\end{rcode}
\[
\begin{aligned}
S(a)&=\{1+e^{-k(a-a_{50})}\}^{-1},\quad k>0,\\
\log F_{t,a}&=\beta_F+\log S(a)+\eta^F_{t,a}.
\end{aligned}
\]
\micro{\texttt{mu\_form = \string~ mono(age)} instead adds cumulative nonnegative log-mean steps. Both curves require IID/AR1; annual F can depart from the rising mean.}
\end{mathbox}\vspace{2mm}
\begin{mathbox}{Natural mortality}
\begin{rcode}
M_settings = list(process = "iid",
  mu_supplied = ~ M0)
\end{rcode}
\[
\log M_{t,a}=\log M^0_{t,a}+\eta^M_{t,a}.
\]
\micro{\texttt{M0} comes from \texttt{pop\_covariates}. M states share age blocks and start in year 2 by default; other cells retain the mean. \texttt{"off"} fixes M to its mean; \texttt{mu\_form} adds log-mean effects.}
\end{mathbox}\vspace{2mm}
\begin{mathbox}{Variation in N, F and M}
\begin{rcode}
F_settings = list(process = "cor_rw",
  sd_form = ~ factor(age))
\end{rcode}
\micro{Ordinary age-based SD formulas give $\sigma_a=\exp(X_a\beta_\sigma)$. N, F and M have separate parameters; shared SDs do not share states.}
\[
\begin{aligned}
\text{IID: }&\eta_{t,a}\sim\mathcal N(0,\sigma_a^2),\\
\text{RW: }&\eta_{t,a}-\eta_{t-1,a}\sim\mathcal N(0,\sigma_a^2).
\end{aligned}
\]
\micro{AR1 stationary covariance (state/block distances):}
\[
\operatorname{Cov}(\eta_{t,a},\eta_{s,b})=
\frac{\sigma_a\sigma_b\phi_t^{|t-s|}\phi_a^{|a-b|}}
{(1-\phi_t^2)(1-\phi_a^2)}.
\]
\micro{F \texttt{"cor\_rw"}: independent annual increment vectors with}
\[
\Sigma_{ab}=\sigma_a\sigma_b\rho^{|a-b|},\quad-1<\rho<1.
\]
\micro{RW starts are unpenalized. N/M \texttt{"off"} gives $\eta=0$.}
\end{mathbox}
\end{minipage}\hfill
\begin{minipage}[t]{.325\linewidth}\vspace{0pt}
\begin{mathbox}{Survey catchability}
\begin{rcode}
index_settings = list(
  q_form = ~ mono(q_block), q_link = "log")
\end{rcode}
\[
\eta_{q,k}=\beta_q+\sum_{j<k}d_j,\quad d_j\ge0,
\qquad q_k=e^{\eta_{q,k}}.
\]
\micro{Zero steps give exact plateaus; \texttt{\string~ q\_block} leaves age patterns unrestricted. With \texttt{q\_link = "logit"}:}
\[
q_k=\{1+e^{-\eta_{q,k}}\}^{-1}.
\]
\micro{\texttt{\string~ logistic(age)} multiplies inverse-link q by $S(a)$. \texttt{iid()}, \texttt{rw()} and \texttt{ar1()} can add Gaussian variation. q covariates come from \texttt{index}.}
\end{mathbox}\vspace{2mm}
\begin{mathbox}{Catch and survey observations}
\begin{rcode}
catch_settings = list(sd_form = ~ 1)
index_settings = list(sd_form = ~ survey)
\end{rcode}
\[
\begin{aligned}
\widetilde C_{t,a}&=N_{t,a}\frac{F_{t,a}}{Z_{t,a}}(1-e^{-Z_{t,a}}),\\
\widetilde I_i&=q_iN_{t_i,a_i}e^{-s_iZ_{t_i,a_i}},\\
\log Y_i&\sim\mathcal N(\log\widetilde Y_i,\sigma_i^2),\\
\sigma_i&=\sigma_i^0\exp(X_{sd,i}\beta_{sd}).
\end{aligned}
\]
\micro{$s_i$: survey sampling fraction. $\sigma_i^0=1$ unless a baseline SD is provided. Gaussian formula effects can also modify log SD. Predictions are medians; zero/NA observations are excluded, with complete predictions.}
\end{mathbox}\vspace{2mm}
\smallkey{Check fit, uncertainty and predictive skill}\par\vspace{1mm}
\begin{rcode}
check_tam(fit)
hind <- fit_hindcast(fit, folds = 20)
hind$hindcast_rmse
\end{rcode}
\micro{The joint likelihood includes active process, formula-effect and observation densities. RTMB integrates random effects by Laplace approximation; intervals use local curvature. Standardized residuals $(\log Y_i-\log\widetilde Y_i)/\sigma_i$ depend on fitted, shrunken states. Convergence does not establish identifiability. Hindcasts return log RMSE and retrospective Mohn's $\rho$.}
